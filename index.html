⚡ AvatarController.js 嵌入代码

只需复制以下 JS 类，在您的对话 App 中通过 avatar.setEmotion('THINKING') 即可轻松嵌入视频同款动态表情！ 

// 1. 引入并初始化
import { AvatarRenderer } from './js/avatar-renderer.js';

const canvas = document.getElementById('myAvatarCanvas');
const avatar = new AvatarRenderer(canvas);

// 2. 根据 NLP 对话判别结果改变表情
avatar.setEmotion('THINKING');  // 触发 3D 轨道彩虹环旋转 + 圆角三角变体
avatar.setEmotion('WRITING');   // 触发粒子流与脉冲
avatar.setEmotion('ANGRY');     // 倾斜眼神 + 压扁震荡
avatar.setEmotion('HAPPY');     // 弹簧跳跃 + 弯月笑眼
avatar.setEmotion('IDLE');      // 恢复平缓呼吸与眨眼
